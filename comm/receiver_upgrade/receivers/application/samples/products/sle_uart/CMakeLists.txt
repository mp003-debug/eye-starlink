set(eye_dir ${CMAKE_CURRENT_SOURCE_DIR}/eye_link)

foreach(unit eye_main eye_radio eye_endpoint eye_fanout eye_event eye_share)
  list(APPEND eye_sources ${eye_dir}/${unit}.c)
endforeach()

  list(APPEND eye_sources ${eye_dir}/eye_home.c ${CMAKE_CURRENT_SOURCE_DIR}/care_hmi.c)
else()
  list(APPEND eye_sources ${eye_dir}/eye_wearable.c ${CMAKE_CURRENT_SOURCE_DIR}/wearable_epd.c ${eye_dir}/eye_share_render.c)
endif()
set(eye_headers ${eye_dir} ${CMAKE_CURRENT_SOURCE_DIR})
set(SOURCES ${SOURCES} ${eye_sources} PARENT_SCOPE)
set(PUBLIC_HEADER ${PUBLIC_HEADER} ${eye_headers} PARENT_SCOPE)
